class Trainer:
    def __init__(self, args, accelerator=None, dont_build_optimizer=False, run_id=None):
        # set up the trainer
        self.args = args
        if not accelerator:
            self.accelerator = Accelerator(
                log_with="wandb",
                gradient_accumulation_steps=self.args.accumulate_steps
            )
        else:
            self.accelerator = accelerator
        self.plot, self.get_plots = plot_logger(accelerator=self.accelerator, args=args)

        try:
            AcceleratorState().deepspeed_plugin.deepspeed_config["train_micro_batch_size_per_gpu"]  = self.args.batch_size
        except AttributeError:
            pass
        self.accelerator.init_trackers(
            project_name="thoughtbubbles", 
            config=vars(args),
            init_kwargs={"wandb": {"mode": None if args.wandb else "disabled",
                                   "resume": "allow",
                                   "id": run_id,
                                   "settings": wandb.Settings(init_timeout=12000),
                                   "name": args.experiment}},
        )

        # ...and the output path
        save_dir = Path(args.out_dir) / args.experiment
        save_dir.mkdir(parents=True, exist_ok=True)

        self.save_dir = str(save_dir / "checkpoint")
        self.best_dir = str(save_dir / "best")

        # create config from arguments
        self.config = ThoughtGPTConfig(
            block_size=args.block_size,
            vocab_size=args.vocab_size,
            n_layer=args.n_layer,
            n_head=args.n_head,
            n_embd=args.n_embd,
            dropout=args.dropout,
            upproj_size=(vars(args).get("upproj_size", 4)),
            bias=args.bias,
            max_thought_budget=args.max_thought_budget,
            fork_map=(vars(args).get("fork_map", "static")),
            pos_embeds=(vars(args).get("pos_embeds", "learn")),
            detach_scores=vars(args).get("detach_scores", False),
            rope_type=vars(args).get("rope_type", "soft"),
            layer_plan=vars(args).get("plan", ["fork" for _ in range(args.n_layer)]),
            expand_size=vars(args).get("expand_size", 4),
            openrightmost=vars(args).get("openrightmost", False),
        )

        # set up the model
        self.model = ThoughtbubbleGPT(self.config)

        # optimizer
        if not dont_build_optimizer:
            self.optim = self.model.configure_optimizers(
                weight_decay=args.weight_decay,
                learning_rate=args.lr,
                betas=(args.beta1, args.beta2),
                device_type="cuda" if torch.cuda.is_available() else "cpu"
            )
            self.scheduler = get_cosine_schedule_with_warmup(
                self.optim,
                int((self.args.total_steps*self.args.warmup_pct)//self.args.accumulate_steps),
                int(self.args.total_steps//self.args.accumulate_steps)
            )

        # compute training size + the counter (useful for mid-checkpoint recovery) 
        self.total_batches = self.args.total_steps
        self.global_step_counter_ = 0
        self.best_val_score_ = float("-inf") # "score" means higher is better 

        # weeeeeeeeeeee
        if not dont_build_optimizer:
            (self.model, self.optim, self.scheduler) = self.accelerator.prepare(self.model, self.optim, self.scheduler)
        else:
            (self.model) = self.accelerator.prepare(self.model)
        if self.accelerator.is_main_process and args.wandb:
            wandb.watch(self.model)

    def _vomit(self):
        """recursively vomit every method and attribute of self into a namespace

        only useful if you are Jack and has a weird Jupyter setup. I apologise
        for the this abuse of all that's good about Python.
        """

        from types import SimpleNamespace
        ns = SimpleNamespace()

        from torch.nn import ModuleList
        buffer = [(i, self) for i in dir(self) if i[0] != "_"]
        names = []
        while len(buffer) > 0:
            member, head = buffer.pop(-1)
            attr = getattr(head, member)
            ns.__dict__[member] = attr

            def include(attr):
                for j in dir(attr):
                    if j not in names and j[0] != "_":
                        buffer.append((j, attr))
                        names.append(j)

            # some special rules for including things 
            if isinstance(attr, ModuleList):
                for component in attr:
                    include(component)
            else:
                include(attr)

        return ns

    def train(self):
        self.epoch()
        self.finish()

    def finish(self):
        self.accelerator.end_training()

    def batch(self, slice="train", deterministic_key=None):

        x,y = get_batch(self.args, self.config, slice,
                        self.device, deterministic_key=deterministic_key)
        mask = (~(x == self.args.ignore_idx)
                    if self.args.ignore_idx != None
                    else None)
        return x.long(),y.long(),mask

    def val(self):
        self.model.eval()
        losses = []
        scores = []
        count = 0
        for i in range(0, vars(self.args).get("validation_steps", 256), self.args.batch_size):
            if i % self.args.report_interval == 0 and self.accelerator.is_main_process:
                logger.info("VAL | {}/{}", i+1, vars(self.args).get("validation_steps", 256))
            with torch.inference_mode():
                x,y,mask = self.batch("val", deterministic_key=i)
                output, loss = self.model(x,y, args=self.args)
                losses.append(self.gather(loss).cpu().item())
                scores.append(((1/loss)*x.size(0)).item()) # because we save higher scores
                count += x.size(0)

        loss = sum(losses)/len(losses)
        score = sum(scores)/count
        metrics = { "val/loss": loss, "val/score": score }
        self.model.train()

        return score, metrics

    def epoch(self):
        if self.accelerator.is_main_process:
            logger.info("BEGIN EPOCH")

        # because sometimes the load function may skip some epochs
        gsc = self.global_step_counter_ # copy to a static number, since we'll be incrementing it
        for indx in range(gsc, self.total_batches):
            # take a step
            with torch.autocast(device_type=self.device.type, dtype=torch.bfloat16):
                if indx % self.args.plot_interval == 0 and indx != 0 and self.global_step_counter_ % self.args.accumulate_steps == 0:
                    with self.plot(self.global_step_counter_//self.args.accumulate_steps,
                                   debug=(not self.args.wandb)):
                        loss, train_metrics = self.step(indx)
                else:
                    loss, train_metrics = self.step(indx)
            train_metrics["train/lr"] = self.optim.param_groups[0]["lr"]

            # perform logging, and then increment
            # (we do this because global_step_counter_
            #  is useful not as the # of steps but how
            #  many we need to skip for warm start)
            if indx % self.args.report_interval == 0 and indx != 0:
                if self.global_step_counter_ % self.args.accumulate_steps == 0:
                    self.accelerator.log(train_metrics, step=self.global_step_counter_//self.args.accumulate_steps)
                if self.accelerator.is_main_process:
                    logger.info("TRAIN | {}/{} | loss {}",
                                self.global_step_counter_, self.total_batches, loss)
            self.global_step_counter_ += 1

            logger.debug("STEP | {} | {}", indx, train_metrics)

            # save a checkpoint, if needed
            if (indx % self.args.checkpoint_interval == 0 and indx != 0 and
                self.accelerator.is_main_process):
                self.save(self.save_dir)
            # perform validation and save a checkpoint, if needed
            if indx % self.args.validation_interval == 0 and indx != 0:
                with torch.autocast(device_type=self.device.type, dtype=torch.bfloat16):
                    score, val_metrics = self.val()
                    # self.viz()
                self.accelerator.log(val_metrics, step=self.global_step_counter_//self.args.accumulate_steps)
                if self.accelerator.is_main_process:
                    logger.info("VAL | {} | score {}", self.global_step_counter_, score)

                if score > self.best_val_score_ and self.accelerator.is_main_process:
                    logger.info("VAL | BEST SCORE | score {}", score)
                    self.best_val_score_ = score
                    self.save(self.best_dir)

    def step(self, indx):
        x,y,mask = self.batch("train")
        if indx % self.args.accumulate_steps == 0:

            if "memorize" in self.args.data_dir:
                _, loss = self.model(x,y, dont_merge=True, args=self.args)
            elif "aqua" in self.args.data_dir.lower():
                _, loss = self.model(x[:, :-1], targets=x[:, 1:], expanded_targets=y, padding_mask=~mask, dont_merge=True, args=self.args)
            elif "gsm8k" in self.args.data_dir.lower():
                _, loss = self.model(x, targets=mask, expanded_targets=y, padding_mask=(x == 0), dont_merge=True, args=self.args)
            else:
                _, loss = self.model(x,y, args=self.args)
            self.accelerator.backward(loss/self.args.accumulate_steps)
            self.optim.step()
            self.scheduler.step()
            self.optim.zero_grad()
        else:
            with self.accelerator.no_sync(self.model):
                if "memorize" in self.args.data_dir:
                    _, loss = self.model(x,y, dont_merge=True, args=self.args)
                elif "aqua" in self.args.data_dir.lower():
                    _, loss = self.model(x[:, :-1], targets=x[:, 1:], expanded_targets=y,
                                         padding_mask=~mask, dont_merge=True, args=self.args)
                elif "gsm8k" in self.args.data_dir.lower():
                    _, loss = self.model(x, targets=mask, expanded_targets=y, padding_mask=(x == 0), dont_merge=True, args=self.args)
                else:
                    _, loss = self.model(x,y, args=self.args)
                self.accelerator.backward(loss/self.args.accumulate_steps)

        loss = self.gather(loss).cpu().item() 
        metrics = { "train/loss": loss }

        return loss, metrics

    def load(self, path):
        self.accelerator.load_state(path)
        with open(os.path.join(path, "config.json"), 'r') as df:
            data = json.load(df)

        self.args = Namespace(**data.get("config", {}))
        self.global_step_counter_ = data.get("steps", 0)
        self.best_val_score_ = data.get("score", 0)

    def save(self, path):
        logger.debug("CHECKPOINT | saving checkpoint at {}", path)
        self.accelerator.save_state(path)
        with open(os.path.join(path, "config.json"), 'w') as df:
            output = {
                "config": vars(self.args),
                "steps": self.global_step_counter_,
                "score": self.best_val_score_,
                "wandb": wandb.run.id if self.args.wandb else None
            }
            json.dump(output, df)

    @classmethod
    def from_pretrained(cls, path, disable_wandb=True, accelerator=None, dont_build_optimizer=False):
        with open(os.path.join(path, "config.json"), 'r') as df:
            data = json.load(df)
        args = Namespace(**data.get("config", {}))
        args.wandb = False if disable_wandb else args.wandb
        new = cls(args, accelerator, dont_build_optimizer=dont_build_optimizer, run_id=data.get("wandb"))

        new.args = Namespace(**data.get("config", {}))
        new.global_step_counter_ = data.get("steps", 0)
        new.best_val_score_ = data.get("score", 0)

        from safetensors import safe_open
        tensors = {}
        with safe_open(os.path.join(path, "model.safetensors"),
                       framework="pt", device=0) as f:
            for k in f.keys():
                tensors[k] = f.get_tensor(k)
        # handle weight tying
        tensors["lm_head.weight"] = tensors["transformer.wte.weight"]
        new.model.load_state_dict(tensors)

        if disable_wandb:
            new.args.wandb = False

        return new

    @classmethod
    def from_checkpoint(cls, path, disable_wandb=True, accelerator=None, dont_build_optimizer=False):
        with open(os.path.join(path, "config.json"), 'r') as df:
            data = json.load(df)
        args = Namespace(**data.get("config", {}))
        args.wandb = False if disable_wandb else args.wandb
        new = cls(args, accelerator, dont_build_optimizer=dont_build_optimizer, run_id=data.get("wandb"))
        new.load(path)

        if disable_wandb:
            new.args.wandb = False

        return new

    @property
    def device(self):
        return self.accelerator.device

    def gather(self, n):
        result = self.accelerator.gather(n)
        if isinstance(result, list):
            return sum(result)/len(result)
        else:
            return result.mean()
